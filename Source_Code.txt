package com.ute.cuuhotulanh

import android.os.Bundle
import androidx.activity.enableEdgeToEdge
import androidx.appcompat.app.AppCompatActivity
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat
import androidx.fragment.app.Fragment
import com.ute.cuuhotulanh.databinding.ActivityMainBinding
import com.ute.cuuhotulanh.ui.FavoritesFragment
import com.ute.cuuhotulanh.ui.FridgeFragment
import com.ute.cuuhotulanh.ui.home.HomeFragment
import com.ute.cuuhotulanh.ui.RecipeDetailFragment
import com.ute.cuuhotulanh.ui.RecipeListFragment

class MainActivity : AppCompatActivity() {

    private lateinit var binding: ActivityMainBinding

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        binding = ActivityMainBinding.inflate(layoutInflater)
        setContentView(binding.root)

        ViewCompat.setOnApplyWindowInsetsListener(binding.root) { v, insets ->
            val bars = insets.getInsets(WindowInsetsCompat.Type.systemBars())
            v.setPadding(bars.left, bars.top, bars.right, bars.bottom)
            insets
        }

        if (savedInstanceState == null) showFragment(HomeFragment())

        binding.bottomNav.setOnItemSelectedListener { item ->
            when (item.itemId) {
                R.id.nav_home -> showFragment(HomeFragment())
                R.id.nav_fridge -> showFragment(FridgeFragment())
                R.id.nav_favorite -> showFragment(FavoritesFragment())
                else -> return@setOnItemSelectedListener false
            }
            true
        }
    }

    private fun showFragment(fragment: Fragment) {
        supportFragmentManager.beginTransaction()
            .replace(R.id.fragment_container, fragment)
            .commit()
    }
}

package com.ute.cuuhotulanh.ui

import android.app.DatePickerDialog
import android.os.Bundle
import android.view.View
import android.widget.ArrayAdapter
import android.widget.AutoCompleteTextView
import android.widget.ImageButton
import android.widget.Toast
import androidx.core.content.ContextCompat
import androidx.fragment.app.Fragment
import com.google.android.material.button.MaterialButton
import com.google.android.material.textfield.TextInputEditText
import com.ute.cuuhotulanh.R
import java.util.Calendar

class AddIngredientFragment : Fragment(R.layout.fragment_add_ingredient) {

    private var selectedCategory: String = "Rau củ"

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        val btnBack = view.findViewById<ImageButton>(R.id.btnBack)
        val etIngredientName = view.findViewById<TextInputEditText>(R.id.etIngredientName)
        val etQuantity = view.findViewById<TextInputEditText>(R.id.etQuantity)
        val autoUnit = view.findViewById<AutoCompleteTextView>(R.id.autoUnit)
        val etExpiryDate = view.findViewById<TextInputEditText>(R.id.etExpiryDate)
        val btnSave = view.findViewById<MaterialButton>(R.id.btnSaveIngredient)

        btnBack.setOnClickListener { parentFragmentManager.popBackStack() }

        val units = listOf("Quả", "Gram", "Kg", "Lít", "Hộp", "Bó")
        val adapter = ArrayAdapter(requireContext(), android.R.layout.simple_dropdown_item_1line, units)
        autoUnit.setAdapter(adapter)
        autoUnit.setText("Quả", false)

        val calendar = Calendar.getInstance()
        etExpiryDate.setOnClickListener {
            val year = calendar.get(Calendar.YEAR)
            val month = calendar.get(Calendar.MONTH)
            val day = calendar.get(Calendar.DAY_OF_MONTH)

            DatePickerDialog(
                requireContext(),
                { _, selectedYear, selectedMonth, selectedDay ->
                    val formatted = "$selectedDay/${selectedMonth + 1}/$selectedYear"
                    etExpiryDate.setText(formatted)
                },
                year, month, day
            ).show()
        }

        val categoryButtons = listOf(
            R.id.btnCategoryVegetable to "Rau củ",
            R.id.btnCategoryMeat to "Thịt",
            R.id.btnCategoryWater to "Nước",
            R.id.btnCategoryEgg to "Trứng",
            R.id.btnCategoryFruit to "Trái cây",
            R.id.btnCategoryFish to "Cá",
            R.id.btnCategoryDry to "Đồ khô",
            R.id.btnCategoryOther to "Khác"
        )

        categoryButtons.forEach { (buttonId, categoryName) ->
            val button = view.findViewById<MaterialButton>(buttonId)
            button.setOnClickListener {
                selectedCategory = categoryName
                updateCategorySelection(view, categoryButtons)
            }
        }

        updateCategorySelection(view, categoryButtons)

        btnSave.setOnClickListener {
            val name = etIngredientName.text?.toString()?.trim().orEmpty()
            val quantity = etQuantity.text?.toString()?.trim().orEmpty()
            val expiry = etExpiryDate.text?.toString()?.trim().orEmpty()

            if (name.isBlank() || quantity.isBlank() || expiry.isBlank()) {
                Toast.makeText(requireContext(), "Vui lòng nhập đầy đủ thông tin", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            Toast.makeText(
                requireContext(),
                "Đã thêm nguyên liệu: $name ($selectedCategory)",
                Toast.LENGTH_SHORT
            ).show()

            parentFragmentManager.popBackStack()
        }
    }

    private fun updateCategorySelection(view: View, categoryButtons: List<Pair<Int, String>>) {
        categoryButtons.forEach { (buttonId, categoryName) ->
            val button = view.findViewById<MaterialButton>(buttonId)
            val isSelected = categoryName == selectedCategory

            button.isSelected = isSelected
            button.setBackgroundTintList(
                ContextCompat.getColorStateList(
                    requireContext(),
                    if (isSelected) R.color.green_light else R.color.white
                )
            )
            button.setTextColor(
                ContextCompat.getColor(
                    requireContext(),
                    if (isSelected) R.color.green_dark else R.color.gray_dark
                )
            )
            button.strokeColor = ContextCompat.getColorStateList(requireContext(), R.color.gray_light)
        }
    }
}

package com.ute.cuuhotulanh.ui

import android.view.LayoutInflater

import android.view.ViewGroup

import androidx.recyclerview.widget.RecyclerView
import com.ute.cuuhotulanh.R
import com.ute.cuuhotulanh.model.Ingredient

class IngredientAdapter(
    private val ingredients: List<Ingredient>
) : RecyclerView.Adapter<IngredientAdapter.IngredientViewHolder>() {

    class IngredientViewHolder(itemView: View) : RecyclerView.ViewHolder(itemView) {
        val ivIcon: ImageView = itemView.findViewById(R.id.ivIngredientIcon)
        val tvName: TextView = itemView.findViewById(R.id.tvIngredientName)
        val tvMeta: TextView = itemView.findViewById(R.id.tvIngredientMeta)
        val tvUseSoon: TextView = itemView.findViewById(R.id.tvUseSoon)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): IngredientViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_ingredient, parent, false)
        return IngredientViewHolder(view)
    }

    override fun onBindViewHolder(holder: IngredientViewHolder, position: Int) {
        val ingredient = ingredients[position]

        holder.ivIcon.setImageResource(ingredient.imageRes)
        holder.tvName.text = ingredient.name
        holder.tvMeta.text = "${ingredient.quantity} ${ingredient.unit} · ${ingredient.expiry}"

        if (ingredient.shouldUseSoon) {
            holder.tvUseSoon.visibility = View.VISIBLE
        } else {
            holder.tvUseSoon.visibility = View.GONE
        }
    }

    override fun getItemCount(): Int = ingredients.size
}

package com.ute.cuuhotulanh.ui

import android.os.Bundle
import android.view.View
import android.widget.TextView
import androidx.fragment.app.Fragment
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.google.android.material.button.MaterialButton
import com.ute.cuuhotulanh.R
import com.ute.cuuhotulanh.util.FakeData
import com.ute.cuuhotulanh.util.StateLayout
import com.ute.cuuhotulanh.util.UiState

class RecipeListFragment : Fragment(R.layout.fragment_recipe_list) {

    private lateinit var stateLayout: StateLayout
    private lateinit var recyclerView: RecyclerView
    private lateinit var tvResultCount: TextView
    private lateinit var btnFewIngredients: MaterialButton
    private lateinit var btnQuick: MaterialButton

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        view.findViewById<View>(R.id.btnBack).setOnClickListener {
            parentFragmentManager.popBackStack()
        }

        recyclerView = view.findViewById(R.id.rvRecipes)
        tvResultCount = view.findViewById(R.id.tvResultCount)
        btnFewIngredients = view.findViewById(R.id.chipFewIngredients)
        btnQuick = view.findViewById(R.id.chipQuick)
        recyclerView.layoutManager = LinearLayoutManager(requireContext())

        // Dòng "Với trứng, cà chua, thịt cá,..." theo nguyên liệu nhận từ Home
        val ingredients = arguments?.getStringArrayList("ingredients").orEmpty()
        view.findViewById<TextView>(R.id.tvIngredientsSummary).text = when {
            ingredients.isEmpty() -> "Với nguyên liệu trong tủ lạnh của bạn"
            ingredients.size <= 3 -> "Với " + ingredients.joinToString(", ")
            else -> "Với " + ingredients.take(3).joinToString(", ") + ",..."
        }

        stateLayout = StateLayout(view, recyclerView) {
            // TODO: gọi lại API khi bấm "Thử lại" (tạm giả lập tải lại)
            stateLayout.render(UiState.LOADING)
            view.postDelayed({ applyFilters() }, 800)
        }
        stateLayout.setEmptyText(
            title = "Chưa tìm thấy món phù hợp",
            message = "Thử thêm nguyên liệu hoặc bỏ bớt bộ lọc nhé"
        )

        // Bấm nút lọc để bật/tắt, danh sách cập nhật ngay
        btnFewIngredients.addOnCheckedChangeListener { _, _ -> applyFilters() }
        btnQuick.addOnCheckedChangeListener { _, _ -> applyFilters() }

        applyFilters()

        // --- DEBUG: nhấn giữ tiêu đề để xem lần lượt các trạng thái. Xóa khi đã nối API thật ---
        var debugIndex = UiState.CONTENT.ordinal
        view.findViewById<View>(R.id.tvTitle).setOnLongClickListener {
            debugIndex = (debugIndex + 1) % UiState.values().size
            stateLayout.render(UiState.values()[debugIndex])
            true
        }
        // --- HẾT DEBUG ---
    }

    /**
     * Lọc theo dữ liệu giả:
     * - "Nấu nhanh": chỉ giữ món có thời gian <= 15 phút
     * - "Ít nguyên liệu nhất": sắp xếp món thiếu ít nguyên liệu lên trước
     */
    private fun applyFilters() {
        var recipes = FakeData.recipes()

        if (btnQuick.isChecked) {
            recipes = recipes.filter { it.time <= 15 }
        }
        if (btnFewIngredients.isChecked) {
            recipes = recipes.sortedBy { missingCount(it.ingredientMiss) }
        }

        // showFavorite = true: thiết kế có tim đỏ trên mỗi thẻ ở màn này
        recyclerView.adapter = RecipeResultAdapter(
            recipes = recipes,
            showFavorite = true
        ) { recipeId ->
            openRecipeDetail(recipeId)
        }

        tvResultCount.text = "Tìm thấy ${recipes.size} món phù hợp"
        stateLayout.render(if (recipes.isEmpty()) UiState.EMPTY else UiState.CONTENT)
    }

    /** "Thiếu 2" -> 2 */
    private fun missingCount(text: String): Int =
        text.filter { it.isDigit() }.toIntOrNull() ?: 0

    private fun openRecipeDetail(id: Int) {
        val fragment = RecipeDetailFragment()
        fragment.arguments = Bundle().apply { putInt("id", id) }

        parentFragmentManager.beginTransaction()
            .replace(R.id.fragment_container, fragment)
            .addToBackStack(null)
            .commit()
    }
}

package com.ute.cuuhotulanh.ui.home

import android.os.Bundle

import com.ute.cuuhotulanh.ui.RecipeDetailFragment
import com.ute.cuuhotulanh.ui.RecipeListFragment

class HomeFragment : Fragment(R.layout.fragment_home) {

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        val listRecipe: List<Recipe> = listOf(
            Recipe(
                1,
                "Trứng xào cà chua",
                R.drawable.img_food,
                13,
                "Có 3 nguyên liệu",
                "Có 3 nguyên liệu"
            ),
            Recipe(
                2,
                "Bún đậu mắm tôm",
                R.drawable.img_food,
                13,
                "Có 3 nguyên liệu",
                "Có 3 nguyên liệu"
            ),
            Recipe(
                3,
                "Bánh mì chảo",
                R.drawable.img_food,
                13,
                "Có 3 nguyên liệu",
                "Có 3 nguyên liệu"
            ),
            Recipe(
                4,
                "Bánh mì chảo",
                R.drawable.img_food,
                13,
                "Có 3 nguyên liệu",
                "Có 3 nguyên liệu"
            ),
        )
        val adapter = RecipeAdapter(listRecipe) { selectedId ->
            navigateToRecipeDetail(selectedId)
        }
        view.findViewById<RecyclerView>(R.id.rvSuggestions).adapter = adapter

        setupIngredientInput(view)

        // Nút "Tìm món ăn": chuyển sang màn Gợi ý tìm món ăn, kèm danh sách nguyên liệu
        view.findViewById<View>(R.id.btnSearch).setOnClickListener {
            navigateToRecipeList(getIngredients(view))
        }
    }

    /** Gõ tên nguyên liệu rồi bấm dấu "+" (hoặc phím Done) để thêm thành chip; bấm X trên chip để xóa. */
    private fun setupIngredientInput(view: View) {
        val inputLayout = view.findViewById<TextInputLayout>(R.id.textInputLayoutIngredients)
        val editText = view.findViewById<TextInputEditText>(R.id.editTextIngredients)
        val chipGroup = view.findViewById<ChipGroup>(R.id.chipGroupIngredients)

        fun addChip() {
            val name = editText.text?.toString()?.trim().orEmpty()
            if (name.isEmpty()) return

            val exists = (0 until chipGroup.childCount).any {
                (chipGroup.getChildAt(it) as Chip).text.toString().equals(name, ignoreCase = true)
            }
            if (!exists) {
                val chip = Chip(requireContext()).apply {
                    text = name
                    isCloseIconVisible = true
                    setOnCloseIconClickListener { chipGroup.removeView(this) }
                }
                chipGroup.addView(chip)
            }
            editText.text?.clear()
        }

        inputLayout.setEndIconOnClickListener { addChip() }
        editText.imeOptions = EditorInfo.IME_ACTION_DONE
        editText.setOnEditorActionListener { _, actionId, _ ->
            if (actionId == EditorInfo.IME_ACTION_DONE) {
                addChip()
                true
            } else {
                false
            }
        }
    }

    private fun getIngredients(view: View): List<String> {
        val chipGroup = view.findViewById<ChipGroup>(R.id.chipGroupIngredients)
        return (0 until chipGroup.childCount).map {
            (chipGroup.getChildAt(it) as Chip).text.toString()
        }
    }

    private fun navigateToRecipeDetail(id: Int) {
        val fragment = RecipeDetailFragment()

package com.ute.cuuhotulanh.util

import android.view.View
import android.widget.TextView
import com.google.android.material.button.MaterialButton
import com.ute.cuuhotulanh.R

/** Các trạng thái hiển thị của một màn danh sách. */
enum class UiState { CONTENT, LOADING, EMPTY, ERROR }

/**
 * Điều khiển layout_state.xml (đã include vào màn hình).
 *
 * @param root    view gốc của màn hình (view truyền vào onViewCreated)
 * @param content view hiển thị khi có dữ liệu (thường là RecyclerView)
 * @param onRetry được gọi khi bấm nút "Thử lại" ở trạng thái lỗi
 *
 * Dùng show() / gone() trong Extensions.kt để bật tắt từng khối.
 */
class StateLayout(
    root: View,
    private val content: View,
    onRetry: () -> Unit = {}
) {
    private val stateRoot: View = root.findViewById(R.id.stateRoot)
    private val loading: View = root.findViewById(R.id.layoutLoading)
    private val empty: View = root.findViewById(R.id.layoutEmpty)
    private val error: View = root.findViewById(R.id.layoutError)

    private val tvEmptyTitle: TextView = root.findViewById(R.id.tvEmptyTitle)
    private val tvEmptyMessage: TextView = root.findViewById(R.id.tvEmptyMessage)
    private val tvErrorTitle: TextView = root.findViewById(R.id.tvErrorTitle)
    private val tvErrorMessage: TextView = root.findViewById(R.id.tvErrorMessage)

    init {
        root.findViewById<MaterialButton>(R.id.btnRetry).setOnClickListener { onRetry() }
    }

    /** Đổi nội dung chữ của trạng thái rỗng (vd: màn Yêu thích). */
    fun setEmptyText(title: String, message: String) {
        tvEmptyTitle.text = title
        tvEmptyMessage.text = message
    }

    /** Đổi nội dung chữ của trạng thái lỗi nếu cần. */
    fun setErrorText(title: String, message: String) {
        tvErrorTitle.text = title
        tvErrorMessage.text = message
    }

    /** Chỉ một khối được hiện tại một thời điểm. */
    fun render(state: UiState) {
        if (state == UiState.CONTENT) content.show() else content.gone()
        if (state == UiState.LOADING) loading.show() else loading.gone()
        if (state == UiState.EMPTY) empty.show() else empty.gone()
        if (state == UiState.ERROR) error.show() else error.gone()

        // Không để lớp phủ chặn thao tác chạm vào danh sách khi đang hiện nội dung
        if (state == UiState.CONTENT) stateRoot.gone() else stateRoot.show()
    }
}
